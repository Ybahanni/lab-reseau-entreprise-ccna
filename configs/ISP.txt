! ============================================
! ISP - Routeur simulant le fournisseur d'accès
! La loopback 8.8.8.8 simule une destination Internet
! ============================================
enable
configure terminal
hostname ISP
no ip domain-lookup
!
interface GigabitEthernet0/0
 description Vers client R1
 ip address 203.0.113.1 255.255.255.252
 no shutdown
!
interface Loopback0
 description Simulation Internet
 ip address 8.8.8.8 255.255.255.255
!
! Aucune route vers les reseaux prives du client :
! seul le trafic translate (NAT) peut revenir.
!
end
write memory
