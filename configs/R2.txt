! ============================================
! R2 - Routeur du site distant (Cisco 2911)
! DHCP local, OSPF, SSH
! ============================================
enable
configure terminal
hostname R2
no ip domain-lookup
ip domain-name lab.local
enable secret CHANGE_ME
username admin privilege 15 secret CHANGE_ME
service password-encryption
banner motd # Acces reserve au personnel autorise #
!
interface GigabitEthernet0/0
 description LAN site distant
 ip address 192.168.40.1 255.255.255.0
 no shutdown
!
interface GigabitEthernet0/1
 description Lien vers R1
 ip address 10.0.12.2 255.255.255.252
 no shutdown
!
ip dhcp excluded-address 192.168.40.1 192.168.40.10
ip dhcp pool LAN_DISTANT
 network 192.168.40.0 255.255.255.0
 default-router 192.168.40.1
 dns-server 8.8.8.8
!
router ospf 1
 router-id 2.2.2.2
 network 192.168.40.0 0.0.0.255 area 0
 network 10.0.12.0 0.0.0.3 area 0
 passive-interface GigabitEthernet0/0
!
! Executer ensuite : crypto key generate rsa  (taille 2048)
ip ssh version 2
line console 0
 login local
 logging synchronous
line vty 0 4
 transport input ssh
 login local
 exec-timeout 10 0
!
end
write memory
